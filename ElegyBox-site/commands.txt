# ElegyBox generation commands
# Defaults: melody_strength=2.0, melody_sigma=1.5, phrase_arc=4.0, phrase_bars=4
#           key=random, chord_key_strength=1.0, enforce_cadence=True
#           lh_pattern=none, lh_bass_split=58, lh_vel=3
#           melody_boost=2, accomp_reduce=1
# Only non-default values are listed per command.
# Stepwise/lyrical styles use tight melody defaults (sigma=1.5, strength=2.0).
# Wide/arpeggio styles override: melody_sigma 5.0-6.0, melody_strength 0.5.
# key=none disables diatonic chord bias (impressionist/chromatic styles).
# chord_key_strength is soft: lower values keep more of the model's own chords.
# enforce_cadence now only resolves the ending; phrase-level harmony stays model-driven.
# lh_pattern choices:
#   alberti : 8th-note broken chord, root,5th,3rd,5th × 2 per bar
#   stride  : long bass (dotted quarter) + staccato chord cluster on off-beats
#   murky   : sustained root + soft octave punch on beats 2 & 4
#   waltz   : long root + short chord on beats 2 & 3
#   walking : stepwise quarter-note line through chord/scale tones, chromatic approach to next chord
#   none    : default; preserves the model's own bass writing
# lh_bass_split=64 for Romantic/dense styles with thick mid-register voicings.
